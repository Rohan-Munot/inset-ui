## Overview

The Checkbox component allows users to toggle a boolean option on or off. It features a smooth check animation and accessible markup.

## Installation

```bash
bun add @inset/ui
```

## Usage

```tsx
import { Checkbox } from '@inset/ui/checkbox';

export default function Example() {
  return (
    <Checkbox />
  );
}
```

## With Label

```tsx
import { Checkbox } from '@inset/ui/checkbox';
import { Label } from '@inset/ui/label';

export default function Example() {
  return (
    <div className="flex items-center gap-2">
      <Checkbox id="terms" />
      <Label htmlFor="terms">Accept terms and conditions</Label>
    </div>
  );
}
```

## Controlled

```tsx
import { useState } from 'react';
import { Checkbox } from '@inset/ui/checkbox';

export default function Example() {
  const [checked, setChecked] = useState(false);

  return (
    <Checkbox
      checked={checked}
      onCheckedChange={setChecked}
    />
  );
}
```

## Indeterminate

```tsx
<Checkbox checked="indeterminate" />
```

## States

### Disabled

```tsx
<Checkbox disabled />
<Checkbox disabled checked />
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `checked` | `boolean \| 'indeterminate'` | - | Controlled checked state |
| `defaultChecked` | `boolean` | `false` | Initial checked state |
| `onCheckedChange` | `(checked: boolean) => void` | - | Called when state changes |
| `disabled` | `boolean` | `false` | Disables the checkbox |
| `required` | `boolean` | `false` | Makes the checkbox required |
